cmake_minimum_required(VERSION 3.20)
project(snet_consumer LANGUAGES CXX)
if(NOT DEFINED SNET_SOURCE_DIR)
    message(FATAL_ERROR "Pass -DSNET_SOURCE_DIR=/absolute/path/to/SimpleNet")
endif()
set(BUILD_TESTING OFF CACHE BOOL "No library tests in this consumer" FORCE)
set(SIMPLENET_BUILD_EXAMPLES OFF CACHE BOOL "No library examples in this consumer" FORCE)
set(SIMPLENET_BUILD_DOCS OFF CACHE BOOL "No library documentation in this consumer" FORCE)
add_subdirectory("${SNET_SOURCE_DIR}" snet)
add_executable(snet_consumer_check main.cpp)
target_link_libraries(snet_consumer_check PRIVATE SimpleNet)
